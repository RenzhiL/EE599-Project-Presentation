% \deckpart{file}{section title}{presenter}: a section, its cover slide, then the owner's file (used in main.tex).
% In a solo build (solo_<name>.tex) the sections not being built are left out entirely, cover included.
\makeatletter
\newcommand{\sectioncover}[1]{%
  \begin{frame}[plain, noframenumbering]
    \vfill
    {\usebeamerfont{section title}\LARGE\color{usccardinal}\insertsection\par}
    \vspace{0.6em}{\color{uscgold}\rule{0.6\linewidth}{1.2pt}}\par\vspace{0.6em}
    {\large\color{uscgray}#1\par}
    \vfill
  \end{frame}}
\newcommand{\deckpart}[3]{%
  \ifdefined\onlypart
    \edef\deck@list{\onlypart}\edef\deck@file{#1}%
    \expandafter\expandafter\expandafter\in@\expandafter\expandafter\expandafter{\expandafter\deck@file\expandafter}\expandafter{\deck@list}%
  \else\in@true\fi
  \ifin@\section{#2}\sectioncover{#3}\include{#1}\fi}
% \backuppart{file}{title}: backup slides after \appendix (used in main.tex): a cover slide, then the owner's file.
% Not in the contents; not counted in the frame total.
\newcommand{\backuppart}[2]{%
  \ifdefined\onlypart
    \edef\deck@list{\onlypart}\edef\deck@file{#1}%
    \expandafter\expandafter\expandafter\in@\expandafter\expandafter\expandafter{\expandafter\deck@file\expandafter}\expandafter{\deck@list}%
  \else\in@true\fi
  \ifin@
    \begin{frame}[plain, noframenumbering]
      \vfill
      {\usebeamerfont{section title}\LARGE\color{usccardinal}#2\par}
      \vspace{0.6em}{\color{uscgold}\rule{0.6\linewidth}{1.2pt}}\par
      \vfill
    \end{frame}
    \include{#1}%
  \fi}
\makeatother
